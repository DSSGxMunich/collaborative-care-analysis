\begin{frame}[fragile]{How to run any of these}
  Every slide carries a snippet that loads the real study file and shows the
  problem as a count or a range. Nothing prints individual patient records.
  \vspace{6pt}

  Run them from the repository root. Paste this preamble first, once:
  \vspace{4pt}
  \begin{Verbatim}[fontsize=\tiny, frame=single, framesep=3pt]
uv run python
import warnings; warnings.filterwarnings("ignore")
import importlib
import pandas as pd
from collaborative_care_analysis.config import RAW_DATASETS_DIR

def loader(name):
    return importlib.import_module("collaborative_care_analysis.data_loading." + name)

def load(name):
    return loader(name).load()

def label_of(study_dir, column):
    """Find the statistical file that holds `column` and return its label."""
    import pyreadstat
    d = RAW_DATASETS_DIR / study_dir
    for f in sorted(list(d.rglob("*.dta")) + list(d.rglob("*.sav"))):
        read = pyreadstat.read_dta if f.suffix == ".dta" else pyreadstat.read_sav
        try:
            _, meta = read(str(f), metadataonly=True)
        except Exception:
            continue
        lab = meta.column_names_to_labels.get(column)
        if lab:
            return f.name, lab
    return None, None
  \end{Verbatim}
  \vspace{4pt}
  {\footnotesize Then paste any slide's snippet. \ans{Status} on each slide says
   whether the pipeline already deals with the problem.}

born = pd.to_datetime(t0["Gebdatum"], errors="coerce")
real = 2005 - born.dt.year
