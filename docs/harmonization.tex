\begin{frame}{The loaders: what was wrong, what was fixed}
  \pause
  All 30 studies reviewed against their own papers and codebooks.
  \vspace{8pt}
  \pause
  \begin{itemize}\footnotesize
    \setlength{\itemsep}{5pt}
    \item \textbf{An age that was not an age.} One study's age came from a two
          digit birth year used directly. \ans{Wrong by up to 70 years.}
    \pause
    \item \textbf{Two studies with swapped treatment arms.} Intervention and
          control the wrong way round. \ans{2{,}270 rows.}
    \pause
    \item \textbf{Every "no" turned into "unknown".} Comorbidity flags where a
          blank means no. Read wrong, \ans{a thousand healthy patients vanish.}
    \pause
    \item \textbf{Two studies silently dropped out.} Two authors' names contain
          an umlaut. A file copy rewrote it invisibly. \ans{No error message.}
  \end{itemize}
  \vspace{8pt}
  \pause
  \begin{center}\scriptsize
  \begin{tabular}{ll}
    \toprule
    reshaped & ds\_29, which was returning its raw file unreshaped \\
    made to fail loudly & missing zips, bare asserts, silent row drops \\
    dtypes & blanks to missing \emph{before} types are inferred \\
    documented & which cohort each export actually contains \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{6pt}
  \pause
  {\footnotesize 47 problems recorded with what was done about each, each on
   its own branch so each can be reviewed on its own.}
\end{frame}

\begin{frame}{Reading the paperwork pays}
  \pause
  Twenty columns in one study are labelled only "Ltc-1" to "Ltc-20".
  Its own codebook calls them \ans{"unclear"}.
  \vspace{10pt}
  \pause
  The paper says patients had \ans{"a mean of 6.2 (SD 3.0) long term
  conditions other than diabetes or heart disease"}.
  \vspace{10pt}
  \pause
  Counting the columns that are not zero gives 7.49. Subtracting each
  patient's diabetes and heart disease gives \ans{6.35}.
  \vspace{10pt}
  \pause
  Twenty unusable columns became one usable number, from one sentence in a
  results section.
\end{frame}

\begin{frame}{How we know it is right}
  \pause
  Wherever a paper prints a number, we recompute it from our data and compare.
  \vspace{10pt}
  \pause
  \begin{center}\footnotesize
  \begin{tabular}{lll}
    \toprule
    what we counted & in which study & does it match the paper \\
    \midrule
    four illnesses & ds\_23 & \ans{yes, patient for patient} \\
    on antidepressants & ds\_26, ds\_09, ds\_08 & \ans{yes, all three} \\
    lives alone & ds\_11 & \ans{yes} \\
    average age & 15 studies & 11 exact, worst off by 0.7y \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{10pt}
  \pause
  \ans{192 automated tests}: value ranges, encodings, types, missing keys,
  attributes that must not change between visits, and statistical package
  missing codes read as numbers.
  \vspace{6pt}
  \pause
  {\footnotesize Where a paper reports nothing, we say the mapping is
   unvalidated rather than implying it is checked.}
\end{frame}

\begin{frame}{What we refused to harmonize}
  \pause
  \begin{itemize}
    \setlength{\itemsep}{8pt}
    \item \textbf{Education as a ladder.} A German school certificate is not a
          US college degree. Only "finished secondary school" survives.
    \pause
    \item \textbf{Ethnicity.} National category systems that do not map, and
          the usual shortcut is meaningless for the Indian cohort.
    \pause
    \item \textbf{Two depression questionnaires.} Same construct, different
          scales, and no patient ever took both.
    \pause
    \item \textbf{Hospital admission counts.} Counted over different periods
          and for different reasons in each study.
  \end{itemize}
  \vspace{8pt}
  \pause
  {\footnotesize Each written down with its reason, so nobody rediscovers it.}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────
\begin{frame}{What is still outstanding}
  \pause
  \textbf{1. Who each trial let in.}
  \vspace{4pt}
  \begin{itemize}\footnotesize
    \item Inclusion and exclusion criteria read from the papers, \ans{238 of
          them, none yet verified by a human}.
    \item They decide which patients a result can be applied to, and they
          explain a zero: not measured, or excluded by design.
  \end{itemize}
  \vspace{12pt}
  \pause
  \textbf{2. What each trial actually did.}
  \vspace{4pt}
  \begin{itemize}\footnotesize
    \item "Collaborative care" is not one intervention. Case manager,
          supervision, medication support, psychological therapy, stepped care:
          each trial combines them differently.
    \item These components are \ans{only in the papers}, not in any dataset.
    \item Without them the comparison is thirty labels reading "intervention".
  \end{itemize}
\end{frame}
